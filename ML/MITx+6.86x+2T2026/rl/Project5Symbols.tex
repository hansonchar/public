\section*{Summary}

\begin{itemize}[leftmargin=*, itemsep=0.5em]
    \item Full record of interaction:
    \[
    (h_{0}, s_{0}, a_{0}, b_{0}, r_{0}, \dots, h_{t}, s_{t}, a_{t}, b_{t}, r_{t}, h_{t+1}, \dots)
    \]

    \item Observalbe record of interaction:
    \[
    (s_{0}, a_{0}, b_{0}, r_{0}, \dots, s_{t}, a_{t}, b_{t}, r_{t}, \dots)
    \]

    \item Total discounted reward:
    \[
    \sum_{t=0}^\infty \gamma^t r_t
    \]

    \item Expected cumulative discounted reward:
    \[
    \mathbb{E} \left[ \sum_{t=0}^{\infty} \gamma^{t} R(h_{t}, a_{t}, b_{t}) \;\middle|\; (a_{t}, b_{t}) \sim \pi \right]
    \]

    \item Maximum expected total discounted reward starting from a given observable state $s$ over all possible policies:
    \[
    V^*(s) = \max_{\pi} \mathbb{E} \left[ \sum_{t=0}^{\infty} \gamma^{t} R(h_{t}, a_{t}, b_{t}) \;\middle|\; h_{0} = h(s),\, s_{0} = s,\, (a_{t}, b_{t}) \sim \pi \right]
    \]

    \item The weighted average of $V^*(s)$ over all possible starting states:
    \[
    V^* = \mathbb{E}_{h_0 \sim \Gamma_{0},\, s \sim \Psi(h)} \left[ V^*(s) \right]
    \]

    \item Maximum expected total discounted reward from a given observable starting state $s$, taking the specific action $(a,b)$ at state $s$ at time $t=0$, over all possible policies.
    \[
    Q^*(s, a, b) = \max_{\pi} \mathbb{E} \left[ \sum_{t=0}^{\infty} \gamma^{t} R(h_{t}, a_{t}, b_{t}) \;\middle|\; h_{0} = h(s),\, s_{0} = s,\, a_{0} = a,\, b_{0} = b,\, (a_{t}, b_{t}) \sim \pi \text{ for } t \ge 1 \right]
    \]

    \item Optimal policy/command at state $s$
    \[
    \pi^*(s) = \arg\max_{(a,b) \in C} Q^*(s, a, b)
    \]
\end{itemize}

\section*{Clarifying notes}

\begin{itemize}[itemsep=0.3em]
    \item $s_t$ is sampled from $h_t$ via $\Psi(h_t)$ at every time step. So even if $h_t = h_{t+1},$ $s_t$ may not equal $s_{t+1}.$
    \item The reason of $(a_t,b_t) \sim \pi$ is because even though the policy function $\pi$ is deterministic given $s_t,$ the observable state $s_t$ itself is randomly generated from $h_t$ via the stochastic function $\Psi,$ and $h_t$ itself is randomly generated from the transition probability function $P$ if $t > 0$ or $\Gamma_0$ if $t=0.$
    \item $\Gamma_0$ and $P$ are fixed, but they are not directly observable by the player.
    \item $V^*$ is the weighted average of $V^*(s)$ over all possible starting states --- even though $\Gamma_0$ is uniform, $\Psi$ is not necessarily uniform.
\end{itemize}
